cmake_minimum_required(VERSION 3.21)
project(pitv_native VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.22)
pkg_check_modules(QRENCODE REQUIRED IMPORTED_TARGET libqrencode)
qt_add_executable(pitv-native
    src/main.cpp src/Controller.cpp src/Controller.h src/MediaEngine.cpp src/MediaEngine.h
    src/SseParser.cpp src/SseParser.h src/LyricsLayout.cpp src/LyricsLayout.h
    src/Fonts.cpp src/Fonts.h src/QrProvider.h qml.qrc)
target_link_libraries(pitv-native PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Network
    PkgConfig::GSTREAMER PkgConfig::QRENCODE)
target_compile_options(pitv-native PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS pitv-native RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(pitv-native-tests tests/CoreTests.cpp src/Controller.cpp src/Controller.h src/SseParser.cpp src/LyricsLayout.cpp src/Fonts.cpp)
    target_include_directories(pitv-native-tests PRIVATE src)
    target_link_libraries(pitv-native-tests PRIVATE Qt6::Test Qt6::Gui Qt6::Network)
    target_compile_definitions(pitv-native-tests PRIVATE TEST_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/library.json")
    add_test(NAME native-core COMMAND pitv-native-tests -platform offscreen)
    add_test(NAME native-ui-smoke COMMAND pitv-native --smoke --fixture ${CMAKE_CURRENT_SOURCE_DIR}/tests/library.json -platform offscreen)
    set_tests_properties(native-ui-smoke PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QUICK_BACKEND=software")
    function(add_ui_smoke name route)
        add_test(NAME native-ui-${name} COMMAND pitv-native --smoke --fixture ${CMAKE_CURRENT_SOURCE_DIR}/tests/library.json --route ${route} ${ARGN} -platform offscreen)
        set_tests_properties(native-ui-${name} PROPERTIES TIMEOUT 20 ENVIRONMENT "QT_QUICK_BACKEND=software")
    endfunction()
    add_ui_smoke(articles "0")
    add_ui_smoke(reader "0,0")
    add_ui_smoke(lyrics "1,0")
    add_ui_smoke(categories "2")
    add_ui_smoke(media "2,0")
    add_ui_smoke(playlist "2,0,1")
    add_ui_smoke(player "2,0,0")
    add_ui_smoke(player-error "2,0,0" --simulate-error)
    add_ui_smoke(catalog "3")
    add_ui_smoke(book "3,0")
    add_ui_smoke(faq "4")
    add_ui_smoke(background "5")
    add_ui_smoke(connect "6")
    add_ui_smoke(light "0,0" --light)
endif()
